cmake_minimum_required(VERSION 3.25)
project(CanaryLedger VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(gate STATIC src/gate.cpp)
target_include_directories(gate PUBLIC include)
target_compile_features(gate PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(gate PRIVATE /W4)
else()
    target_compile_options(gate PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(canaryledger src/main.cpp)
target_link_libraries(canaryledger PRIVATE gate)
if(MINGW)
    target_link_options(canaryledger PRIVATE -static)
endif()
install(TARGETS canaryledger RUNTIME DESTINATION bin)
install(FILES policy/release.conf DESTINATION share/canaryledger)

if(BUILD_TESTING)
    add_executable(gate_tests tests/gate_test.cpp)
    target_link_libraries(gate_tests PRIVATE gate)
    add_test(NAME policy_contract COMMAND gate_tests)
endif()
